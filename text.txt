    <div class="container">
        <h1>📖 Poetry Reader</h1>
        <div class="search-box">
            <input type="text" id="searchInput" placeholder="Search by author or poem title...">
            <button onclick="searchPoem()">Search</button>
            <button onclick="getRandomPoem()">Random</button>
        </div>
        <div id="poemContainer"></div>
    </div>

    <script>
        const API_BASE = 'https://poetrydb.org';

        async function searchPoem() {
            const query = document.getElementById('searchInput').value.trim();
            if (!query) {
                document.getElementById('poemContainer').innerHTML = '<div class="error">Please enter a poem title or author name</div>';
                return;
            }

            document.getElementById('poemContainer').innerHTML = '<div class="error">Loading...</div>';

            try {
                let response = await fetch(`${API_BASE}/title/${encodeURIComponent(query)}`);
                let data = await response.json();

                if (data.status === 404) {
                    response = await fetch(`${API_BASE}/author/${encodeURIComponent(query)}`);
                    data = await response.json();
                }

                if (data.status === 404) {
                    document.getElementById('poemContainer').innerHTML = '<div class="error">No poems found. Try another search.</div>';
                    return;
                }

                const poems = Array.isArray(data) ? data : [data];
                displayPoem(poems[0]);
            } catch (error) {
                document.getElementById('poemContainer').innerHTML = '<div class="error">Error: ' + error.message + '</div>';
            }
        }

        async function getRandomPoem() {
            document.getElementById('poemContainer').innerHTML = '<div class="error">Loading...</div>';
            try {
                const response = await fetch(`${API_BASE}/random`);
                const data = await response.json();
                displayPoem(data[0]);
            } catch (error) {
                document.getElementById('poemContainer').innerHTML = '<div class="error">Error: ' + error.message + '</div>';
            }
        }

        function displayPoem(poem) {
            const poemText = Array.isArray(poem.lines) ? poem.lines.join('\n') : poem.lines;
            const html = `
                <div class="poem-display">
                    <div class="poem-title">${poem.title}</div>
                    <div class="poem-author">by ${poem.author}</div>
                    <div class="poem-text">${poemText}</div>
                </div>
            `;
            document.getElementById('poemContainer').innerHTML = html;
        }

        document.getElementById('searchInput').addEventListener('keypress', (e) => {
            if (e.key === 'Enter') searchPoem();
        });
    </script>
